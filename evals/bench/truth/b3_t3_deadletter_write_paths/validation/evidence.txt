def run_pipeline(raw):
    """Push one raw payload through every stage in `STAGE_ORDER`.

    `raw` may be a plain dict (a brand new event, run from `ingest`
    onward) or an `Event` already in flight (a retry, resumed from
    `transform` onward since only `ingest` needs the original raw dict).

    Returns the final `Event` on success, or `None` if the event was
    dead-lettered - either immediately, or after its retries were
    exhausted.
    """
    if isinstance(raw, Event):
        event = raw
        stages = STAGE_ORDER[1:]
    else:
        event = STAGE_DISPATCH["ingest"](raw)
        stages = STAGE_ORDER[1:]

    for stage_name in stages:
        stage_fn = STAGE_DISPATCH[stage_name]
        try:
            event = stage_fn(event)
        except RelayError as exc:
            if retry.should_retry(exc, event.attempts):
                event.bump_attempt()
                delay = retry.backoff_for(event.attempts)
                scheduler.schedule_retry(event, run_pipeline, delay)
                state.mark_retried()
                logger.info(
                    "scheduled retry %s/%s for %s after %s",
                    event.attempts,
                    config.MAX_RETRIES,
                    event.event_id,
                    type(exc).__name__,
                )
            else:
                deadletter.write_dead_letter(event, exc)
                state.mark_dead_lettered()
            return None

    state.mark_processed(event.event_id)
    return event

     1	"""A minimal in-process scheduler for retried events.
     2	
     3	`schedule_retry` stores a callback alongside the event it belongs to and
     4	returns immediately; the callback runs later, from `drain_ready`, whenever
     5	something chooses to call it. `expire_stale` is the scheduler's other exit
     6	door: a retry that has been waiting too long for its turn is given up on
     7	directly, without ever running its callback.
     8	"""
     9	from __future__ import annotations
    10	
    11	import logging
    12	import time
    13	from dataclasses import dataclass
    14	from typing import Callable
    15	
    16	from relay.errors import RetryExpiredError
    17	from relay.sinks import deadletter, log_sink
    18	
    19	logger = logging.getLogger(__name__)
    20	
    21	
    22	@dataclass
    23	class _ScheduledRetry:
    24	    event: object
    25	    run_again: Callable[[object], object]
    26	    ready_at: float
    27	    scheduled_at: float
    28	
    29	
    30	class Scheduler:
    31	    def __init__(self) -> None:
    32	        self._pending: list[_ScheduledRetry] = []
    33	
    34	    def schedule_retry(self, event, run_again: Callable[[object], object], delay_s: float) -> None:
    35	        """Queue `run_again(event)` to run once `delay_s` has elapsed."""
    36	        now = time.time()
    37	        self._pending.append(
    38	            _ScheduledRetry(event=event, run_again=run_again, ready_at=now + delay_s, scheduled_at=now)
    39	        )
    40	        log_sink.log_retry_scheduled(event, delay_s)
    41	
    42	    def pending_count(self) -> int:
    43	        return len(self._pending)
    44	
    45	    def drain_ready(self, now: float | None = None) -> int:
    46	        """Invoke every scheduled callback whose delay has elapsed.
    47	
    48	        Returns the number of callbacks invoked. A callback failing
    49	        unexpectedly is logged and skipped rather than allowed to stop
    50	        the rest of the batch from draining - `run_again` owns deciding
    51	        what a failure of its own event means (retry again, dead-letter,
    52	        or something else) and normally handles that itself.
    53	        """
    54	        now = time.time() if now is None else now
    55	        ready: list[_ScheduledRetry] = []
    56	        still_pending: list[_ScheduledRetry] = []
    57	        for item in self._pending:
    58	            (ready if item.ready_at <= now else still_pending).append(item)
    59	        self._pending = still_pending
    60	
    61	        for item in ready:
    62	            try:
    63	                item.run_again(item.event)
    64	            except Exception:  # noqa: BLE001 - one bad scheduled retry shouldn't block its siblings
    65	                logger.exception("scheduled retry callback failed for %s", item.event.event_id)
    66	        return len(ready)
    67	
    68	    def expire_stale(self, max_pending_s: float, now: float | None = None) -> int:
    69	        """Give up on any retry that has been sitting in the queue for
    70	        longer than `max_pending_s`, whether or not its delay has
    71	        elapsed yet. Each one is dead-lettered directly, without ever
    72	        running its callback. Returns the number expired.
    73	        """
    74	        now = time.time() if now is None else now
    75	        cutoff = now - max_pending_s
    76	        expiring: list[_ScheduledRetry] = []
    77	        still_pending: list[_ScheduledRetry] = []
    78	        for item in self._pending:
    79	            (expiring if item.scheduled_at <= cutoff else still_pending).append(item)
    80	        self._pending = still_pending
    81	
    82	        for item in expiring:
    83	            error = RetryExpiredError(
    84	                f"retry for {item.event.event_id} expired after {max_pending_s}s pending"
    85	            )
    86	            deadletter.write_dead_letter(item.event, error)
    87	        return len(expiring)
    88	
    89	
    90	scheduler = Scheduler()
